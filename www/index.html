سكربت SQL لجدول المزامنة في Supabase

انسخ هذا السكربت ونفّذه مرة واحدة في SQL Editor داخل مشروعك على Supabase لتخزين ومزامنة بيانات الجمعية بين المواقع المختلفة:

create table if not exists public.association_state (
  workspace_id text primary key,
  payload jsonb not null,
  updated_at timestamptz not null default now()
);

alter table public.association_state enable row level security;

drop policy if exists "allow_all_association_state" on public.association_state;
create policy "allow_all_association_state"
  on public.association_state
  for all
  using (true)
  with check (true);

alter publication supabase_realtime add table public.association_state;
